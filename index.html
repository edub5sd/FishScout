Apple iMessage Opener function sendViaAppleMessages() { const { text } = getDraftMessage(); const payload = `${text}\n\nSent from Spot Finder: ${window.location.href}`; const isAppleDevice = /iPad|iPhone|iPod|Macintosh/i.test(navigator.userAgent); if (isAppleDevice) { // Works on iPhone, iPad, and Mac computers window.location.href = `sms:&body=${encodeURIComponent(payload)}`; showToast("💬 Opening Apple Messages..."); } else { // Windows PC Fallback copyToClipboardFallback(payload); alert("Apple does not allow iMessage on Windows PCs.\n\nYour message was copied to your clipboard! Use the blue (↑) button to chat inside Spot Finder, or paste it into your paired Phone Link app."); } } // 2. Android SMS Opener (Works on both Phone & PC) function sendViaAndroidSMS() { const { text } = getDraftMessage(); const payload = `${text}\n\nSent from Spot Finder: ${window.location.href}`; const isMobile = /Android|iPhone|iPad|iPod/i.test(navigator.userAgent); if (isMobile) { // Native phone SMS window.location.href = `sms:?body=${encodeURIComponent(payload)}`; showToast("🤖 Opening phone texting app..."); } else { // PC: Copy text and open Google Messages for Web copyToClipboardFallback(payload); showToast("📋 Message copied! Opening Google Messages Web..."); setTimeout(() => { window.open('https://messages.google.com/web', '_blank'); }, 600); } }
